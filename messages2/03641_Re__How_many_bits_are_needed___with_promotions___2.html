Infact, I have just written a bit of C code to check this out. Infact,
there are only 8694 ways to arrange the number of Queens, Knights,
Bishops, Rooks and Pawns, so that can be reduced to 13.5 bits, or 27 bits
for the two sides.

However, I can improve that further. The huffman encoding will be more
efficient by at least 2 bits per piece not on the board, so if I can make
the header smaller for 16 pieces on the board, and larger for smaller
numbers it will all cancel out. A little more code gives the following
table of number of non-king pieces against possibilities :

0 = 1 (0 bits)
1 = 5 (3)
2 = 15 (4)
3 = 35 (6)
4 = 70 (7)
5 = 126 (7)
6 = 210 (8)
7 = 330 (9)
8 = 495 (9)
9 = 714 (10)
10 = 987 (10)
11 = 1287 (11)
12 = 1479 (11)
13 = 1455 (11)
14 = 990 (10)
15 = 495 (9)

So if you first code the number of pieces as follows :

15=1, 14=01, 13=001, 12=00011, 0-11 = the 7 bit binary number for 0 to 11
(e.g. 0000000 to 0001011), and then add on to the end, the number of the
position, then you get 10 bits for 15 pieces, 12 for 14, 14 for 13, 16 for
12, and 18 or less for 11 or less.

p.s. Incase you are interested, here is the C Code.

#include <stdio.h>

void main()
{
int queens,knights,bishops,rooks,pawns;
int pro_queens,pro_knights,pro_bishops,pro_rooks;
int count[16];
int total = 0;
int counter;

for (counter = 0; counter < 16; counter++ )
{
count[ counter ] = 0;
}

for ( queens = 0; queens <= 9; queens++ )
{
for ( knights = 0; knights <=10; knights++ )
{
for ( bishops = 0; bishops <=10; bishops++ )
{
for ( rooks = 0; rooks <=10; rooks++ )
{
for ( pawns = 0; pawns <= 10;
pawns++ )
{
pro_queens = queens - 1;
pro_knights = knights - 2;
pro_bishops = bishops - 2;
pro_rooks = rooks - 2;

if ( pro_queens < 0 )
{
pro_queens = 0;
}

if ( pro_knights < 0 )
{
pro_knights = 0;
}

if ( pro_bishops < 0 )
{
pro_bishops = 0;
}

if ( pro_rooks < 0 )
{
pro_rooks = 0;
}

if ( pro_queens +
pro_knights + pro_bishops + pro_rooks + pawns <= 8 )
{
count[ queens +
knights + bishops + rooks + pawns ]++;
total++;
}
}
}
}
}
}

for (counter = 0; counter < 16; counter++ )
{
printf("%d = %d\n", counter, count[ counter ] );
}

printf("Total = %d\n", total );

getchar();
}
