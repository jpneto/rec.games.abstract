and 2 shots, these values were used:

---------------------
Testing the close combat case where the nazi always have the first turn and
1 shot, these values were used:

AmericanHitChance = 0.90;

AmericanFirerate = 2.0;
NaziHitChance = 0.90;
NaziKillChance = 0.127;
NaziFirerate = 1.0;

Firesequence used: nazi, am, nazi, am .... This is the initial case, giving
80 %. When you changed american firerate to 4 the winning chance becomes 86
%. See programcode B.


A simple average of these cases gives 89 %. This simulation gives as result
89 % also at close distance, indicating a direct relation between winning
chance and firepower at close distance too (which surprices me a bit).

Simulation A.
Simulation of turn by turn. Firesequence is: am, nazi, am, nazi ....

procedure TForm1.Button1Click(Sender: TObject);
const
AmericanHitChance = 0.90;
AmericanKillChance = 0.30;
AmericanFirerate = 4.0; give 92 %
NaziHitChance = 0.90;
NaziKillChance = 0.243;
NaziFirerate = 1.0;
var
turn, shot: Integer;
AmericansAlive, AmericansKilled: Single;
NazisAlive, NazisKilled: Single;
RemainingDuels: Single;
begin
AmericansAlive := 1000.0;
NazisAlive := 1000.0;
RemainingDuels := 1000.0;
for turn := 1 to 6 do
begin
for shot := 1 to Round(AmericanFirerate) do
begin
NazisKilled := RemainingDuels * AmericanHitChance *
AmericanKillChance;
NazisAlive := NazisAlive - NazisKilled;
if NazisAlive < AmericansAlive then
RemainingDuels := NazisAlive
else RemainingDuels := AmericansAlive;
end;
AmericansKilled := RemainingDuels * NaziHitChance *
NaziKillChance * NaziFirerate;
AmericansAlive := AmericansAlive - AmericansKilled;
if NazisAlive < AmericansAlive then
RemainingDuels := NazisAlive
else RemainingDuels := AmericansAlive;
end;
Label1.Caption := FloatToStr( AmericansAlive) + ' ' +
FloatToStr( NazisAlive);

Label2.Caption := 'Winning chance = ' + FloatToStr( 100.0 * (1000 -
NazisAlive) /
((1000.0 - NazisAlive) + (1000.0 - AmericansAlive))) +
' %';
{ Winning chance = 100 * NazisKilled/(NazisKilled + AmericansKilled }
Label3.Caption := 'Odds = ' + FloatToStr( (1000.0 - NazisAlive) /
(1000.0 - AmericansAlive));
{ Winning odds = NazisKilled : AmericansKilled }
end;
